# WhatsApp Automation Manager — runtime dependencies.
#
#   python -m venv .venv
#   .venv\Scripts\python.exe -m pip install -r requirements.txt
#
# Three packages left when the UI-Automation transport did: `uiautomation` and
# `pywin32` drove WhatsApp Desktop's accessibility tree, and `psutil` found the
# WhatsApp.exe window to drive. OpenWA is the transport now, reached over HTTP
# with the standard library, so none of them have anything to do — and with
# them went the last reason this had to run on Windows.

# Primary datastore. The [srv] extra pulls in dnspython, which pymongo REQUIRES
# to resolve a mongodb+srv:// (Atlas) address — without it every Atlas
# connection string fails no matter how correct it is.
pymongo[srv]>=4.6

# Verifies the TLS certificate Atlas presents on images whose OS trust store is
# incomplete. Used when present, skipped when not.
certifi

# The desktop UI. 6.5 is the floor: QStyleHints.colorScheme(), which the
# light/dark theme follows, arrived in that release.
PySide6>=6.5

# .env loading. The parser in wadam/config.py is used if this is absent, so a
# missing optional dependency can never stop the application from starting.
python-dotenv>=1.0
